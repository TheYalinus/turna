find_package(Catch2)
if(CATCH2_FOUND)
    message("catch2 found")
else()
    message("Catch2 not found")
    CPMAddPackage("gh:catchorg/Catch2@3.16.1")
endif()
add_executable(unit_tests unit_tests.cpp)
target_link_libraries(unit_tests PRIVATE Catch2::Catch2WithMain turna::turna nlohmann_json::nlohmann_json)
list(APPEND CMAKE_MODULE_PATH ${Catch2_SOURCE_DIR}/extras)
include(CTest)
include(Catch)
enable_testing()
add_test(NAME UnitTests COMMAND unit_tests)
catch_discover_tests(unit_tests)
